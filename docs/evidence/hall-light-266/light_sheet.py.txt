"""Scratch (#266): Hall lighting trial sheet. Usage: light_sheet.py <before dir> <after dir> <footage dir> <out.jpg>"""
import sys, os
import numpy as np
from PIL import Image, ImageDraw
B, A, F, OUT = sys.argv[1:5]
def lab(i, t):
    d = ImageDraw.Draw(i); d.rectangle([0, 0, 7 * len(t) + 6, 13], fill=(0, 0, 0)); d.text((3, 1), t, fill=(255, 255, 255)); return i
def fit(p, w=None, h=None):
    i = Image.open(p).convert("RGB"); k = w / i.width if w else h / i.height
    return i.resize((int(i.width * k), int(i.height * k)), Image.LANCZOS)
def stack(v, w):
    b, a = lab(fit(f"{B}/{v}.png", w=w), v + " before"), lab(fit(f"{A}/{v}.png", w=w), v + " after")
    s = Image.new("RGB", (w, b.height + a.height + 4), (30, 30, 30)); s.paste(b, (0, 0)); s.paste(a, (0, b.height + 4)); return s
def beside(ims, gap=6):
    s = Image.new("RGB", (sum(i.width for i in ims) + gap * (len(ims) - 1), max(i.height for i in ims)), (30, 30, 30)); x = 0
    for i in ims: s.paste(i, (x, 0)); x += i.width + gap
    return s
def save(rows, out):
    W = max(r.width for r in rows); sheet = Image.new("RGB", (W, sum(r.height + 6 for r in rows)), (30, 30, 30)); y = 0
    for r in rows: sheet.paste(r, (0, y)); y += r.height + 6
    q = 88
    while True:
        sheet.save(out, quality=q, optimize=True)
        if os.path.getsize(out) < 295_000 or q < 40: break
        q -= 4
    k = min(1200 / sheet.width, 1200 / sheet.height, 1.0)
    sheet.resize((int(sheet.width * k), int(sheet.height * k)), Image.LANCZOS).save(out.replace(".jpg", "-view.jpg"), quality=85)
    print(out, sheet.size, os.path.getsize(out), "q", q)
far, arch = stack("3-far", 400), stack("4-arch", 400)
r0 = beside([lab(fit(f"{F}/far.jpg", h=far.height), "footage 26.0 s"), far, lab(fit(f"{F}/arch.jpg", h=arch.height), "footage 178.6 s"), arch]); W = r0.width
pair = lambda v: beside([lab(fit(f"{B}/{v}.png", w=W // 2 - 3), v + " before"), lab(fit(f"{A}/{v}.png", w=W // 2 - 3), v + " after")])
save([r0, pair("6-oblique"), pair("5-follow")], OUT + "/1-end-walls-oblique-follow.jpg")
save([stack("1-west", 1400), stack("2-east", 1400)], OUT + "/2-long-walls-before-above-after.jpg")
save([stack("7-floor", 1400)], OUT + "/3-floor-before-above-after.jpg")
