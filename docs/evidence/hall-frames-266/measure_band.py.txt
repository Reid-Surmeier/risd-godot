"""Frame band of a Hall painting read from footage: the work's own picture is matched to video
frames (SIFT, RANSAC homography), the wall is redrawn square-on at PPM px per metre from the
catalogue canvas size, and the frame's outer edge is the outermost sharp colour step on each side."""
import cv2, numpy as np, json, subprocess, sys, os, glob
WT = "/home/reidsurmeier/orca/workspaces/risd-godot/wt-stairwell/modules/shell/prototype/gallery_walk4/"
MOV = os.path.expanduser("~/risd-godot-ingestion/walkthrough/IMG_6344.MOV")
PPM = 300; M = 0.5  # px per metre, margin drawn round the canvas (m)
works = {w["tag"]: w for w in json.load(open(WT + "works.json"))}
sift = cv2.SIFT_create(8000); bf = cv2.BFMatcher()

def frames(tag, t, span=1.5, fps=4):
    d = f"f/{tag}-{t}"; os.makedirs(d, exist_ok=True)
    if not glob.glob(d + "/*.jpg"):
        subprocess.run(["ffmpeg", "-v", "error", "-ss", str(t - span), "-t", str(2 * span), "-i", MOV,
                        "-vf", f"fps={fps},scale=1620:-2", "-q:v", "2", d + "/%03d.jpg"], check=True)
    return sorted(glob.glob(d + "/*.jpg"))

def rectify(tag, path, ref_kd):
    w = works[tag]; cw, ch = w["canvas_w"], w["canvas_h"]
    ref, (k1, d1) = ref_kd
    img = cv2.imread(path); g = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    k2, d2 = sift.detectAndCompute(g, None)
    if d2 is None: return None
    good = [m for m, n in bf.knnMatch(d1, d2, k=2) if m.distance < 0.72 * n.distance]
    if len(good) < 25: return None
    rh, rw = ref.shape[:2]
    # reference pixel -> rectified pixel (canvas at M..M+size)
    src = np.float32([[(k1[m.queryIdx].pt[0] / rw * cw + M) * PPM, (k1[m.queryIdx].pt[1] / rh * ch + M) * PPM] for m in good])
    dst = np.float32([k2[m.trainIdx].pt for m in good])
    H, mask = cv2.findHomography(dst, src, cv2.RANSAC, 3.0)
    if H is None or mask.sum() < 25: return None
    out = cv2.warpPerspective(img, H, (int((cw + 2 * M) * PPM), int((ch + 2 * M) * PPM)), flags=cv2.INTER_AREA)
    # how oblique: the canvas's four sides in the source frame
    c = cv2.perspectiveTransform(np.float32([[[M * PPM, M * PPM], [(M + cw) * PPM, M * PPM], [(M + cw) * PPM, (M + ch) * PPM], [M * PPM, (M + ch) * PPM]]]), np.linalg.inv(H))[0]
    L = [np.linalg.norm(c[i] - c[(i + 1) % 4]) for i in range(4)]
    return out, int(mask.sum()), (L[1] / L[3], L[0] / L[2]), max(L)

def profile(rect, tag, side):
    """mean colour (Lab) at each offset outward from the canvas edge, over the middle half of the side"""
    w = works[tag]; cw, ch = w["canvas_w"], w["canvas_h"]
    lab = cv2.cvtColor(rect, cv2.COLOR_BGR2LAB).astype(np.float32)
    x0, x1, y0, y1 = int(M * PPM), int((M + cw) * PPM), int(M * PPM), int((M + ch) * PPM)
    n = int(0.46 * PPM)
    if side in "lr":
        ya, yb = int(y0 + 0.25 * (y1 - y0)), int(y0 + 0.75 * (y1 - y0))
        strip = lab[ya:yb, x0 - n:x0][:, ::-1] if side == "l" else lab[ya:yb, x1:x1 + n]
        return cut(np.median(strip, axis=0))
    xa, xb = int(x0 + 0.25 * (x1 - x0)), int(x0 + 0.75 * (x1 - x0))
    strip = lab[y0 - n:y0, xa:xb][::-1] if side == "t" else lab[y1:y1 + n, xa:xb]
    return cut(np.median(strip, axis=1))

def cut(p):
    bad = np.where(p[:, 0] < 4)[0]
    return p[:max(bad[0] - 6, 0)] if len(bad) else p

def edge(p):
    if len(p) < int(0.12 * PPM): return float("nan")
    """outermost sharp step: gradient over 2 cm, outermost peak above 45 % of the largest"""
    k = int(0.01 * PPM)
    g = np.linalg.norm(p[2 * k:] - p[:-2 * k], axis=1)  # g[i] is centred on offset i + k
    lo = int(0.02 * PPM)
    thr = 0.45 * g[lo:].max()
    idx = [i for i in range(lo, len(g) - 1) if g[i] >= thr and g[i] >= g[i - 1] and g[i] >= g[i + 1]]
    return (idx[-1] + k) / PPM if idx else float("nan")

if __name__ == "__main__":
    res = {}
    for arg in sys.argv[1:]:
        tag, t = arg.split("@"); t = float(t)
        ref = cv2.imread(WT + f"detail/{tag}.jpg"); kd = sift.detectAndCompute(cv2.cvtColor(ref, cv2.COLOR_BGR2GRAY), None)
        rows = []
        for i, f in enumerate(frames(tag, t)):
            r = rectify(tag, f, (ref, kd))
            if r is None: continue
            rect, inl, obl, size = r
            b = [edge(profile(rect, tag, s)) for s in "ltrb"]
            rows.append({"t": round(t - 1.5 + i / 4, 2), "inl": inl, "obl": [round(float(o), 3) for o in obl], "px": int(size), "band": [round(float(x), 3) for x in b], "f": f})
        rows.sort(key=lambda r: (not (0.85 < r["obl"][0] < 1.18 and 0.85 < r["obl"][1] < 1.18), -r["inl"]))
        res[arg] = rows
        print(tag, works[tag]["acc"], "frames matched", len(rows))
        for r in rows[:4]: print("  t=%6.2f inl=%4d side px=%4d obl(h,w)=%s band l/t/r/b=%s" % (r["t"], r["inl"], r["px"], r["obl"], r["band"]))
        if rows:
            top = np.array([r["band"] for r in rows[:6]]); print("  median of top", len(top), np.round(np.nanmedian(top, axis=0), 3), "spread", np.round(np.nanmax(top, axis=0) - np.nanmin(top, axis=0), 3))
    json.dump(res, open("bands.json", "w"), indent=1)
