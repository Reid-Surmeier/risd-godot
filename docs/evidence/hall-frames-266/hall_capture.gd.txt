## Scratch (#266): the Main Hall from fixed cameras, with the game's renderer. Args: <out dir>
extends SceneTree

const W := 10.0
const L := 26.3
var walk


func _initialize() -> void:
	call_deferred("_run")


func _run() -> void:
	var out: String = OS.get_cmdline_user_args()[0]
	DirAccess.make_dir_recursive_absolute(out)
	root.size = Vector2i(1600, 400)
	walk = load("res://modules/shell/prototype/collection_reconstruction/main_build_walk.gd").new()
	walk.size = Vector2(root.size)
	root.add_child(walk)
	for i in 240:
		await process_frame
	var box: SubViewportContainer = walk._vp.get_parent()
	# the visitor stands between the benches, follow view, so nothing is cut away
	walk._new_action()
	walk._target = null
	walk._held.clear()
	walk._velocity = Vector3.ZERO
	walk._pos = Vector3(0, 0, -13.0)
	walk._last_pos = walk._pos
	walk._space = "gallery"
	walk.view_mode = 2
	walk.view_yaw = 0.0
	walk._yaw = 0.0
	walk.set_process(false)
	walk._kid.pose(0.0, false, 0.0, Vector3(0, 0, -1), 0.0)
	for settle in 12:
		walk._update_camera(1.0)
		await process_frame
	var cam: Camera3D = walk._cam
	print("MASK ", cam.cull_mask, " lit ", walk._baked_lighting)
	var rows := []
	for p in walk._paintings:
		rows.append({"tag": p.tag, "c": [p.center.x, p.center.y, p.center.z], "outer": [p.outer.x, p.outer.y], "n": [p.normal.x, p.normal.z]})
	var f := FileAccess.open(out.path_join("paintings.json"), FileAccess.WRITE)
	f.store_string(JSON.stringify(rows))
	f.close()
	# what the saved bake holds for each work: the box of its frame meshes and of its canvas
	var saved: Node3D = load(OS.get_cmdline_user_args()[1] if OS.get_cmdline_user_args().size() > 1 else "res://modules/shell/prototype/gallery_walk4/baked/room.tscn").instantiate()
	var boxes := {}
	for mesh in saved.find_children("*", "MeshInstance3D", true, false):
		var material := mesh.material_override as StandardMaterial3D
		if material == null or material.albedo_texture == null:
			continue
		var path: String = material.albedo_texture.resource_path
		var kind := "frame" if path.contains("/gallery_walk4/frames/") else ("canvas" if path.contains("/gallery_walk4/canvas/") else "")
		if kind == "":
			continue
		var key := path.get_file().get_basename().replace("-shaped", "") + ":" + kind
		var aabb: AABB = mesh.transform * mesh.mesh.get_aabb()
		boxes[key] = aabb.merge(boxes[key]) if boxes.has(key) else aabb
	var saved_rows := {}
	for key in boxes:
		var b: AABB = boxes[key]
		saved_rows[key] = [b.position.x, b.position.y, b.position.z, b.size.x, b.size.y, b.size.z]
	var sf := FileAccess.open(out.path_join("baked.json"), FileAccess.WRITE)
	sf.store_string(JSON.stringify(saved_rows))
	sf.close()
	saved.free()
	# name, size px, eye, target, ortho size (0 = perspective), fov
	var shots := [
		["5-follow", Vector2i(960, 640), null, null, 0.0, 0.0],
		["1-west", Vector2i(1600, 400), Vector3(3.6, 3.1, -L / 2), Vector3(-5, 3.1, -L / 2), 6.6, 0.0],
		["2-east", Vector2i(1600, 400), Vector3(-3.6, 3.1, -L / 2), Vector3(5, 3.1, -L / 2), 6.6, 0.0],
		["3-far", Vector2i(1000, 660), Vector3(0, 3.1, -L + 8.0), Vector3(0, 3.1, -L), 6.6, 0.0],
		["4-arch", Vector2i(1000, 660), Vector3(0, 3.1, -8.0), Vector3(0, 3.1, 0), 6.6, 0.0],
		["6-oblique", Vector2i(1200, 800), Vector3(4.2, 4.6, -1.5), Vector3(-3.0, 1.0, -15.0), 0.0, 62.0],
		["7-floor", Vector2i(1600, 620), Vector3(0, 5.6, -L / 2), Vector3(0, 0, -L / 2), 10.2, 0.0],
	]
	# close square-on views, 200 px per metre, of every work
	for p in walk._paintings:
		if true:
			var cs := Vector2(p.rec.canvas_w + 1.5, p.rec.canvas_h + 1.5)
			shots.append(["c-" + p.tag, Vector2i(cs * 200.0), p.center + p.normal * 4.0, p.center, cs.y, 0.0])
	for shot in shots:
		root.size = shot[1]
		walk.size = Vector2(root.size)
		await process_frame
		box.stretch_shrink = 1
		if shot[2] != null:
			cam.position = shot[2]
			if shot[4] > 0.0:
				cam.projection = Camera3D.PROJECTION_ORTHOGONAL
				cam.size = shot[4]
			else:
				cam.projection = Camera3D.PROJECTION_PERSPECTIVE
				cam.fov = shot[5]
			cam.look_at(shot[3], Vector3(1, 0, 0) if shot[0] == "7-floor" else Vector3.UP)
		for settle in 8:
			await process_frame
		await RenderingServer.frame_post_draw
		var image: Image = root.get_texture().get_image()
		image.save_png(out.path_join(shot[0] + ".png"))
		print("CAPTURE ", shot[0], " ", image.get_size())
	quit()
